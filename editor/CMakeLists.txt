if(FOUNDATION_SDK_ROOT)
  execute_process(COMMAND "${Python3_EXECUTABLE}" -B
    "${FOUNDATION_POLICY_SOURCE_ROOT}/tools/sdk_verify.py" "${FOUNDATION_SDK_ROOT}"
    COMMAND_ERROR_IS_FATAL ANY OUTPUT_QUIET)
  file(READ "${FOUNDATION_SDK_ROOT}/sdk.json" metadata)
  string(JSON sdk_system GET "${metadata}" target system)
  if(NOT sdk_system STREQUAL "Linux")
    message(FATAL_ERROR "This --sdk editor route requires a native Linux SDK")
  endif()
  foreach(pair "cxx_compiler|CMAKE_CXX_COMPILER" "sysroot|CMAKE_SYSROOT")
    string(REPLACE "|" ";" parts "${pair}")
    list(GET parts 0 field)
    list(GET parts 1 configured)
    string(JSON declared GET "${metadata}" target ${field})
    file(REAL_PATH "${FOUNDATION_SDK_ROOT}/${declared}" expected)
    file(REAL_PATH "${${configured}}" actual)
    if(NOT actual STREQUAL expected)
      message(FATAL_ERROR "SDK and configured ${configured} differ; select the existing SDK toolchain in a fresh tree")
    endif()
  endforeach()
  set(FOUNDATION_VERIFY_SDK TRUE)
endif()

function(foundation_options target)
  foundation_sdk_runtime(${target})
  target_compile_features(${target} PUBLIC cxx_std_20)
  set_target_properties(${target} PROPERTIES CXX_EXTENSIONS OFF)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /permissive- /utf-8)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
  endif()
  if(FOUNDATION_SANITIZERS)
    if(MSVC OR NOT CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
      message(FATAL_ERROR "Editor sanitizer checks require GCC or Clang")
    endif()
    target_compile_options(${target} PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(${target} PUBLIC -fsanitize=address,undefined)
  endif()
endfunction()
function(gui_warnings target)
  foundation_options(${target})
endfunction()

  add_library(foundation_editor_ui STATIC ui/application.cpp ui/canvas.cpp)

  foundation_options(foundation_editor_ui)

    foundation_gui_native_backend("sdl")

  elseif(FOUNDATION_EDITOR_BACKEND STREQUAL "rev")
    foundation_gui_native_backend("rev")

  else()
    foundation_gui_native_backend("fltk")
    set(toolkit foundation_gui_fltk)
  endif()
  set(editor_host "foundation-editor-${FOUNDATION_EDITOR_BACKEND}")
  add_executable(${editor_host} "hosts/${FOUNDATION_EDITOR_BACKEND}_main.cpp")
  target_include_directories(${editor_host} PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
  target_link_libraries(${editor_host} PRIVATE foundation_editor_ui ${toolkit})

    if(FOUNDATION_EDITOR_BACKEND STREQUAL "framebuffer")

      target_link_options(${editor_host} PRIVATE /MANIFEST:EMBED
