if(FOUNDATION_SDK_ROOT)
  execute_process(COMMAND "${Python3_EXECUTABLE}" -B
    "${FOUNDATION_POLICY_SOURCE_ROOT}/tools/sdk_verify.py" "${FOUNDATION_SDK_ROOT}"
    COMMAND_ERROR_IS_FATAL ANY OUTPUT_QUIET)
  file(READ "${FOUNDATION_SDK_ROOT}/sdk.json" metadata)
  string(JSON sdk_system GET "${metadata}" target system)
  if(NOT sdk_system STREQUAL "Linux")
    message(FATAL_ERROR "This --sdk workbench route requires a native Linux SDK")
  endif()
  foreach(pair "cxx_compiler|CMAKE_CXX_COMPILER" "sysroot|CMAKE_SYSROOT")
    string(REPLACE "|" ";" parts "${pair}")
    list(GET parts 0 field)
    list(GET parts 1 configured)
    string(JSON declared GET "${metadata}" target ${field})
    file(REAL_PATH "${FOUNDATION_SDK_ROOT}/${declared}" expected)
    file(REAL_PATH "${${configured}}" actual)
    if(NOT actual STREQUAL expected)
      message(FATAL_ERROR "SDK and configured ${configured} differ; use its toolchain in a fresh tree")
    endif()
  endforeach()
  set(FOUNDATION_VERIFY_SDK TRUE)
endif()

function(foundation_options target)
  foundation_sdk_runtime(${target})
  target_compile_features(${target} PUBLIC cxx_std_20)
  set_target_properties(${target} PROPERTIES CXX_EXTENSIONS OFF)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /permissive- /utf-8)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
  endif()
  if(FOUNDATION_SANITIZERS)
    if(MSVC OR NOT CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
      message(FATAL_ERROR "Workbench sanitizer checks require GCC or Clang")
    endif()
    target_compile_options(${target} PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(${target} PUBLIC -fsanitize=address,undefined)
  endif()
endfunction()
function(gui_warnings target)
  foundation_options(${target})
endfunction()

# The helper is copied beside the executable's build tree. Python uses only its
# standard library; native hosts invoke this selected interpreter explicitly.
file(GLOB runtime_sources CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/services/*.py")
foreach(source IN LISTS runtime_sources)
  get_filename_component(name "${source}" NAME)
  configure_file("${source}" "${CMAKE_CURRENT_BINARY_DIR}/runtime/${name}" COPYONLY)
endforeach()
configure_file("${FOUNDATION_POLICY_SOURCE_ROOT}/tools/openrouter.py"
  "${CMAKE_CURRENT_BINARY_DIR}/runtime/_openrouter_config.py" COPYONLY)
configure_file("${FOUNDATION_POLICY_SOURCE_ROOT}/tools/process_tree.py"
  "${CMAKE_CURRENT_BINARY_DIR}/runtime/_process_tree.py" COPYONLY)
# Only an absolute location is retained. Credential bytes never enter CMake,
# executable definitions, test fixtures or the copied runtime directory.
execute_process(COMMAND "${Python3_EXECUTABLE}" -c
  "import json, pathlib, sys; pathlib.Path(sys.argv[2]).write_text('OPENROUTER_CONFIG_PATH = ' + json.dumps(sys.argv[1], ensure_ascii=True) + '\\n', encoding='utf-8')"
  "${FOUNDATION_POLICY_SOURCE_ROOT}/.env.openrouter"
  "${CMAKE_CURRENT_BINARY_DIR}/runtime/_runtime_paths.py"
  COMMAND_ERROR_IS_FATAL ANY)
file(GLOB core_sources CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/core/*.cpp")
add_library(tinyenv_workbench_core STATIC ${core_sources}
  "${FOUNDATION_POLICY_SOURCE_ROOT}/editor/platform/project_files.cpp"
  "${FOUNDATION_POLICY_SOURCE_ROOT}/editor/platform/process_runner.cpp")
